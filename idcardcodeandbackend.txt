        function copyElement() {
            if (!selectedElement) {
                showFeedbackNotice("No element selected to copy.");
                return;
            }

            const imgNode = selectedElement.querySelector('img');
            const isImgType = !!imgNode;
            const style = window.getComputedStyle(selectedElement);

                left: selectedElement.style.left,
                top: selectedElement.style.top,
                width: selectedElement.style.width,
                height: selectedElement.style.height,
                baseFontFamily: style.fontFamily,
                baseFontSize: style.fontSize,

                opacity: selectedElement.style.opacity || '1',
                borderRadius: selectedElement.style.borderRadius || '0px',
                textAlign: selectedElement.style.textAlign || 'center',
                shape: selectedElement.getAttribute('data-shape') || 'square',
                panX: selectedElement.getAttribute('data-pan-x') || '50',
                panY: selectedElement.getAttribute('data-pan-y') || '50'
            };

        function pasteElement() {
            if (!copiedElementData) {
                showFeedbackNotice("No element to paste. Copy an element first.");
                return;
            }

            const el = document.createElement('div');
            el.className = 'element';
            el.style.left = copiedElementData.left;
            el.style.top = copiedElementData.top;
            if (copiedElementData.width) el.style.width = copiedElementData.width;
            if (copiedElementData.height) el.style.height = copiedElementData.height;
            el.style.opacity = copiedElementData.opacity;
            el.style.borderRadius = copiedElementData.borderRadius;
            el.style.textAlign = copiedElementData.textAlign;
            el.setAttribute('data-shape', copiedElementData.shape);
            el.setAttribute('data-pan-x', copiedElementData.panX);
            el.setAttribute('data-pan-y', copiedElementData.panY);

            if (copiedElementData.type === 'text') {
                el.style.fontFamily = copiedElementData.baseFontFamily;
                el.style.fontSize = copiedElementData.baseFontSize;
                el.style.color = copiedElementData.baseColor;
                el.innerHTML = `
                    <div class="text-content" contenteditable="true" style="text-align: ${copiedElementData.textAlign};">${copiedElementData.innerPayload}</div>
                    <div class="resize-handle"></div>
                `;
            } else {
                el.classList.add('image-element');
                el.innerHTML = `
                    <img src="${copiedElementData.innerPayload}" style="border-radius:inherit; object-position: ${copiedElementData.panX}% ${copiedElementData.panY}%;" />
                    <div class="resize-handle"></div>
                `;
            }

            initElement(el);
            card.appendChild(el);
            selectElement(el);
            refreshLayersListPanel();
            saveHistoryState();
            pushRealTimePreviewSync();
            showFeedbackNotice("Element pasted successfully!");
        }

        // --- Advanced Alignment Detection System ---
        function checkAlignmentGuides(draggedElement) {
            if (!alignmentGuidesEnabled) {
                hideAllAlignmentGuides();
                return;
            }

            const snapThreshold = 6;
            const cardWidth = card.offsetWidth;
            const cardHeight = card.offsetHeight;

            const draggedRect = {
                left: draggedElement.offsetLeft,
                top: draggedElement.offsetTop,
                right: draggedElement.offsetLeft + draggedElement.offsetWidth,
                bottom: draggedElement.offsetTop + draggedElement.offsetHeight,
                centerX: draggedElement.offsetLeft + draggedElement.offsetWidth / 2,
                centerY: draggedElement.offsetTop + draggedElement.offsetHeight / 2,
                width: draggedElement.offsetWidth,
                height: draggedElement.offsetHeight
            };

            const cardCenterX = cardWidth / 2;
            const cardCenterY = cardHeight / 2;

            let snappedX = null;
            let snappedY = null;
            let showVCenter = false, showHCenter = false;
            let showVLeft = false, showVRight = false, showHTop = false, showHBottom = false;

            if (Math.abs(draggedRect.centerX - cardCenterX) < snapThreshold) {
                snappedX = cardCenterX - draggedElement.offsetWidth / 2;
                showVCenter = true;
            }
            if (Math.abs(draggedRect.centerY - cardCenterY) < snapThreshold) {
                snappedY = cardCenterY - draggedElement.offsetHeight / 2;
                showHCenter = true;
            }

            const otherElements = Array.from(card.querySelectorAll('.element')).filter(el => el !== draggedElement);

            otherElements.forEach(otherEl => {
                const otherRect = {
                    left: otherEl.offsetLeft,
                    top: otherEl.offsetTop,
                    right: otherEl.offsetLeft + otherEl.offsetWidth,
                    bottom: otherEl.offsetTop + otherEl.offsetHeight,
                    centerX: otherEl.offsetLeft + otherEl.offsetWidth / 2,
                    centerY: otherEl.offsetTop + otherEl.offsetHeight / 2
                };

                if (Math.abs(draggedRect.left - otherRect.left) < snapThreshold) {
                    snappedX = otherRect.left;
                    showVLeft = true;
                    document.getElementById('align-guide-v-left').style.left = otherRect.left + 'px';
                }
                if (Math.abs(draggedRect.right - otherRect.right) < snapThreshold) {
                    snappedX = otherRect.right - draggedElement.offsetWidth;
                    showVRight = true;
                    document.getElementById('align-guide-v-right').style.left = otherRect.right + 'px';
                }
                if (Math.abs(draggedRect.centerX - otherRect.centerX) < snapThreshold) {
                    snappedX = otherRect.centerX - draggedElement.offsetWidth / 2;
                    showVCenter = true;
                    document.getElementById('align-guide-v-center').style.left = otherRect.centerX + 'px';
                }

                if (Math.abs(draggedRect.top - otherRect.top) < snapThreshold) {
                    snappedY = otherRect.top;
                    showHTop = true;
                    document.getElementById('align-guide-h-top').style.top = otherRect.top + 'px';
                }
                if (Math.abs(draggedRect.bottom - otherRect.bottom) < snapThreshold) {
                    snappedY = otherRect.bottom - draggedElement.offsetHeight;
                    showHBottom = true;
                    document.getElementById('align-guide-h-bottom').style.top = otherRect.bottom + 'px';
                }
                if (Math.abs(draggedRect.centerY - otherRect.centerY) < snapThreshold) {
                    snappedY = otherRect.centerY - draggedElement.offsetHeight / 2;
                    showHCenter = true;
                    document.getElementById('align-guide-h-center').style.top = otherRect.centerY + 'px';
                }
            });

            if (snappedX !== null) draggedElement.style.left = snappedX + 'px';
            if (snappedY !== null) draggedElement.style.top = snappedY + 'px';

            document.getElementById('align-guide-v-center').style.display = showVCenter ? 'block' : 'none';
            if (showVCenter && snappedX === null) document.getElementById('align-guide-v-center').style.left = cardCenterX + 'px';
            
            document.getElementById('align-guide-h-center').style.display = showHCenter ? 'block' : 'none';
            if (showHCenter && snappedY === null) document.getElementById('align-guide-h-center').style.top = cardCenterY + 'px';
            
            document.getElementById('align-guide-v-left').style.display = showVLeft ? 'block' : 'none';
            document.getElementById('align-guide-v-right').style.display = showVRight ? 'block' : 'none';
            document.getElementById('align-guide-h-top').style.display = showHTop ? 'block' : 'none';
            document.getElementById('align-guide-h-bottom').style.display = showHBottom ? 'block' : 'none';
        }

        // --- HTML to Formatted Text Segments Parser ---
        function parseHTMLToSegments(htmlContent) {
            const tempDiv = document.createElement('div');
            tempDiv.innerHTML = htmlContent;
            
            const segments = [];
            
            function traverse(node, inheritedStyles = {}) {
                if (node.nodeType === Node.TEXT_NODE) {
                    const text = node.textContent;
                    if (text) {
                        segments.push({
                            text: text,
                            ...inheritedStyles
                        });
                    }
                } else if (node.nodeType === Node.ELEMENT_NODE) {
                    const tagName = node.tagName.toLowerCase();
                    const newStyles = {...inheritedStyles};
                    
                    if (tagName === 'b' || tagName === 'strong') newStyles.bold = true;
                    if (tagName === 'i' || tagName === 'em') newStyles.italic = true;
                    if (tagName === 'u') newStyles.underline = true;
                    
                    const style = node.style;
                    if (style) {
                        if (style.fontWeight === 'bold') newStyles.bold = true;
                        if (style.fontWeight === 'normal') newStyles.bold = false;
                        if (style.fontStyle === 'italic') newStyles.italic = true;
                        if (style.fontStyle === 'normal') newStyles.italic = false;
                        if (style.textDecoration === 'underline') newStyles.underline = true;
                        
                        if (style.fontFamily) newStyles.fontFamily = style.fontFamily;
                        if (style.fontSize) newStyles.fontSize = style.fontSize;
                        if (style.color) newStyles.color = style.color;
                    }
                    
                    if (tagName === 'br') {
                        segments.push({ text: '\n' });
                    } else {
                        for (let child of node.childNodes) {
                            traverse(child, newStyles);
                        }
                    }
                }
            }
            
            traverse(tempDiv);
            return segments;
        }

        // --- High Definition Rendering System Engine ---
        function generateHDCanvas(scaleFactor, callback) {
            const priorSelectedElement = selectedElement;
            if (selectedElement) selectedElement.classList.remove('active');

            const origWidth = card.offsetWidth;
            const origHeight = card.offsetHeight;

            const canvas = document.createElement('canvas');
            canvas.width = origWidth * scaleFactor;
            canvas.height = origHeight * scaleFactor;
            
            const ctx = canvas.getContext('2d');
            ctx.scale(scaleFactor, scaleFactor);

            ctx.fillStyle = window.getComputedStyle(cardBg).backgroundColor || "#ffffff";
            ctx.fillRect(0, 0, origWidth, origHeight);

            const drawCardElements = () => {
                const elements = card.querySelectorAll('.element');
                let elementsToProcess = elements.length;

                if (elementsToProcess === 0) {
                    if (priorSelectedElement) priorSelectedElement.classList.add('active');
                    callback(canvas);
                    return;
                }
                
                const onElementDrawn = () => {
                    elementsToProcess--;
                    if (elementsToProcess === 0) {
                        if (priorSelectedElement) priorSelectedElement.classList.add('active');
                        callback(canvas);
                    }
                }

                elements.forEach(el => {
                    ctx.save();
                    ctx.globalAlpha = parseFloat(window.getComputedStyle(el).opacity || 1);

                    const x = el.offsetLeft;
                    const y = el.offsetTop;
                    const w = el.offsetWidth;
                    const h = el.offsetHeight;

                    const imgNode = el.querySelector('img');
                    if (imgNode) {
                        const shape = el.getAttribute('data-shape') || 'square';
                        const panX = parseFloat(el.getAttribute('data-pan-x') || 50);
                        const panY = parseFloat(el.getAttribute('data-pan-y') || 50);
                        
                        const tempImg = new Image();
                        tempImg.crossOrigin = "Anonymous";
                        tempImg.onload = () => {
                            ctx.save();
                            ctx.beginPath();
                            if (shape === 'circle') {
                                ctx.arc(x + w/2, y + h/2, Math.min(w,h)/2, 0, Math.PI*2);
                            } else if (shape === 'rounded') {
                                const radius = parseFloat(window.getComputedStyle(el).borderRadius);
                                ctx.moveTo(x + radius, y);
                                ctx.lineTo(x + w - radius, y); ctx.quadraticCurveTo(x + w, y, x + w, y + radius);
                                ctx.lineTo(x + w, y + h - radius); ctx.quadraticCurveTo(x + w, y + h, x + w - radius, y + h);
                                ctx.lineTo(x + radius, y + h); ctx.quadraticCurveTo(x, y + h, x, y + h - radius);
                                ctx.lineTo(x, y + radius); ctx.quadraticCurveTo(x, y, x + radius, y);
                            } else {
                                ctx.rect(x, y, w, h);
                            }
                            ctx.closePath();
                            ctx.clip();

                            const imgRatio = tempImg.width / tempImg.height;
                            const elementRatio = w / h;
                            let sX = 0, sY = 0, sW = tempImg.width, sH = tempImg.height;

                            if (imgRatio > elementRatio) {
                                sW = tempImg.height * elementRatio;
                                sX = (tempImg.width - sW) * (panX / 100);
                            } else {
                                sH = tempImg.width / elementRatio;
                                sY = (tempImg.height - sH) * (panY / 100);
                            }

                            ctx.drawImage(tempImg, sX, sY, sW, sH, x, y, w, h);
                            ctx.restore();
                            ctx.restore();
                            onElementDrawn();
                        };
                        tempImg.onerror = () => {
                           console.error("Could not load image for rendering:", imgNode.src);
                           onElementDrawn();
                        }
                        tempImg.src = imgNode.src;
                    } else {
                        const textContentNode = el.querySelector('.text-content');
                        const style = window.getComputedStyle(el);

                        const baseFontFamily = style.fontFamily;
                        const baseFontSize = parseInt(style.fontSize);
                        const baseColor = style.color;
                        
                        const alignment = style.textAlign || "center";
                        const paddingLeft = 6;
                        const paddingRight = 6;
                        const paddingTop = 6;
                        
                        const htmlContent = textContentNode.innerHTML;
                        const sanitizedHtml = htmlContent.replace(/<div>/g, '<br>').replace(/<\/div>/g, '').replace(/<p>/g, '<br>').replace(/<\/p>/g, '');
                        const lines = sanitizedHtml.split(/<br\s*\/?>/i);
                        
                        const computedLineHeightOffset = baseFontSize * 1.2;
                        let ongoingYOffset = y + paddingTop;

                        lines.forEach(lineHTML => {
                            const segments = parseHTMLToSegments(lineHTML);
                            if (segments.length === 0 && lineHTML === '') {
                                ongoingYOffset += computedLineHeightOffset;
                                return;
                            }
                            
                            let totalLineWidth = 0;
                            segments.forEach(seg => {
                                ctx.save();
                                const segFontSize = seg.fontSize ? parseInt(seg.fontSize) : baseFontSize;
                                const segFontFamily = seg.fontFamily || baseFontFamily;
                                const fontWeight = seg.bold ? 'bold' : 'normal';
                                const fontStyle = seg.italic ? 'italic' : 'normal';
                                ctx.font = `${fontStyle} ${fontWeight} ${segFontSize}px ${segFontFamily}`;
                                totalLineWidth += ctx.measureText(seg.text).width;
                                ctx.restore();
                            });

                            let currentX = x + paddingLeft;
                            if (alignment === "center") {
                                currentX = x + (w / 2) - (totalLineWidth / 2);
                            } else if (alignment === "right") {
                                currentX = x + w - paddingRight - totalLineWidth;
                            }

                            segments.forEach(seg => {
                                ctx.save();
                                
                                const segFontSize = seg.fontSize ? parseInt(seg.fontSize) : baseFontSize;
                                const segFontFamily = seg.fontFamily || baseFontFamily;
                                const segColor = seg.color || baseColor;
                                const fontWeight = seg.bold ? 'bold' : 'normal';
                                const fontStyle = seg.italic ? 'italic' : 'normal';
                                
                                ctx.fillStyle = segColor;
                                ctx.font = `${fontStyle} ${fontWeight} ${segFontSize}px ${segFontFamily}`;
                                ctx.textBaseline = "top";
                                ctx.textAlign = "left";
                                
                                ctx.fillText(seg.text, currentX, ongoingYOffset);
                                
                                if (seg.underline) {
                                    const textMetrics = ctx.measureText(seg.text);
                                    ctx.strokeStyle = segColor;
                                    ctx.lineWidth = Math.max(1, segFontSize / 14);
                                    ctx.beginPath();
                                    const underlineY = ongoingYOffset + segFontSize;
                                    ctx.moveTo(currentX, underlineY);
                                    ctx.lineTo(currentX + textMetrics.width, underlineY);
                                    ctx.stroke();
                                }
                                
                                currentX += ctx.measureText(seg.text).width;
                                ctx.restore();
                            });

                            ongoingYOffset += computedLineHeightOffset;
                        });

                        ctx.restore();
                        onElementDrawn();
                    }
                });
            };

            if (cardBgImageData) {
                const bgImg = new Image();
                bgImg.crossOrigin = "Anonymous";
                bgImg.onload = () => {
                    ctx.save();
                    ctx.globalAlpha = parseFloat(window.getComputedStyle(cardBg).opacity || 1);
                    ctx.drawImage(bgImg, 0, 0, origWidth, origHeight);
                    ctx.restore();
                    drawCardElements();
                };
                bgImg.onerror = () => {
                    console.error("Could not load background image for rendering:", cardBgImageData);
                    drawCardElements();
                };
                bgImg.src = cardBgImageData;
            } else {
                drawCardElements();
            }
        }

        function saveHistoryState(isInitRun = false) {
            const snapshotStr = compileCurrentStateBlueprint();
            if (undoStack.length > 0 && undoStack[undoStack.length - 1] === snapshotStr) return; 
            undoStack.push(snapshotStr);
            if (undoStack.length > 40) undoStack.shift(); 
            if (!isInitRun) redoStack = []; 
        }

        function undoWorkspaceAction() {
            if (undoStack.length <= 1) return; 
            const activeState = undoStack.pop();
            redoStack.push(activeState);
            applyStateBlueprintStr(undoStack[undoStack.length - 1]);
            pushRealTimePreviewSync();
        }

        function redoWorkspaceAction() {
            if (redoStack.length === 0) return;
            const targetStateStr = redoStack.pop();
            undoStack.push(targetStateStr);
            applyStateBlueprintStr(targetStateStr);
            pushRealTimePreviewSync();
        }

        function initElement(el) {
            let isDragging = false;
            let isResizing = false;
            let startX, startY, startW, startH, startLeft, startTop;

            const textContentNode = el.querySelector('.text-content');
            if (textContentNode) {
                textContentNode.addEventListener('input', () => {

                    isResizing = true;

                    if (e.target.getAttribute('contenteditable') === 'true') return;
                    isDragging = true;
                }
                
                selectElement(el);
                startX = e.clientX;
                startY = e.clientY;
                startW = el.offsetWidth;
                startH = el.offsetHeight;
                startLeft = el.offsetLeft;
                startTop = el.offsetTop;
                
                el.setPointerCapture(e.pointerId);
                e.stopPropagation();
            });

            el.addEventListener('pointermove', (e) => {
                if (!isDragging && !isResizing) return;

                const dx = e.clientX - startX;
                const dy = e.clientY - startY;

                if (isResizing) {
                    el.style.width = Math.max(1, startW + dx) + 'px';
                    el.style.height = Math.max(1, startH + dy) + 'px';
                } else if (isDragging) {
                    el.style.left = (startLeft + dx) + 'px';
                    el.style.top = (startTop + dy) + 'px';
                    checkAlignmentGuides(el);
                }
                
                pushRealTimePreviewSync();
            });

            el.addEventListener('pointerup', (e) => {
                if (isDragging || isResizing) {
                    isDragging = false;
                    isResizing = false;
                    el.releasePointerCapture(e.pointerId);
                    
                    hideAllAlignmentGuides();
                    refreshLayersListPanel();
                    saveHistoryState();
                    pushRealTimePreviewSync();
                }
            });

            if (!selectedElement || selectedElement.querySelector('img')) return;

                if (styleProperty === 'fontFamily') document.execCommand('fontName', false, styleValue);
                if (styleProperty === 'fontSize') document.execCommand('fontSize', false, 7);

                            fontEl.removeAttribute("size");

                    }
                }

            selection.removeAllRanges();

function doGet(e) {

function handleRequest(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    var data = {};
    if (e.postData && e.postData.contents) {
      try {
        data = JSON.parse(e.postData.contents);
      } catch (err) {
        data = e.parameter;
      }
    } else if (e.parameter) {
      data = e.parameter;
    }

    var action = data.action;
    var result = { success: false };

    // Setup required Google Sheet tabs automatically
    var sheets = initializeDatabaseSheets();

    if (action === 'init') {
      result = { success: true, message: 'Database initialized successfully.' };
    } 
    else if (action === 'register') {
      result = registerUser(sheets.users, data.user);
    } 
    else if (action === 'login') {
      result = loginUser(sheets.users, data.credentials);
    } 
    else if (action === 'admin_login') {
      result = verifyAdminLogin(data.credentials);
    }
    else if (action === 'request_otp') {
      result = sendPasswordResetOTP(sheets.users, sheets.otp, data.identifier);
    } 
    else if (action === 'reset_password_otp') {
      result = resetPasswordWithOTP(sheets.users, sheets.otp, data.resetData);
    } 
    else if (action === 'save') {
      result = saveTemplateRecord(sheets.blueprints, data.template, data.userToken);
    } 
    else if (action === 'load') {
      var offset = parseInt(data.offset || 0, 10);
      var limit = parseInt(data.limit || 6, 10);
      result = loadTemplateRecords(sheets.blueprints, data.username, data.isAdmin, offset, limit);
    } 
    else if (action === 'delete') {
      result = deleteTemplateRecord(sheets.blueprints, data.uid, data.username, data.isAdmin);
    } 
    else {
      result = { success: false, message: 'Invalid API action request.' };
    }

    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

/**
 * Registers a new user account in Google Sheets.
 */
function registerUser(sheet, user) {
  if (!user || !user.username || !user.email || !user.password) {
    return { success: false, message: "Missing required registration details." };
  }

  var data = sheet.getDataRange().getValues();
  var usernameLower = String(user.username).trim().toLowerCase();
  var emailLower = String(user.email).trim().toLowerCase();

  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).toLowerCase() === usernameLower) {
      return { success: false, message: "Username already exists. Choose another." };
    }
    if (String(data[i][1]).toLowerCase() === emailLower) {
      return { success: false, message: "Email is already registered. Please login." };
    }
  }

  sheet.appendRow([user.username.trim(), emailLower, user.password, new Date().toISOString()]);
  return { success: true, message: "Account registered successfully!", username: user.username.trim(), email: emailLower };
}

/**
 * Authenticates user login credentials.
 */
function loginUser(sheet, creds) {
  if (!creds || !creds.identifier || !creds.password) {
    return { success: false, message: "Please provide both username/email and password." };
  }

  var data = sheet.getDataRange().getValues();
  var idLower = String(creds.identifier).trim().toLowerCase();

  for (var i = 1; i < data.length; i++) {
    var uName = String(data[i][0]).trim();
    var uEmail = String(data[i][1]).trim();
    var uPass = String(data[i][2]);

    if ((uName.toLowerCase() === idLower || uEmail.toLowerCase() === idLower) && uPass === creds.password) {
      return { success: true, message: "Login successful!", username: uName, email: uEmail };
    }
  }
  return { success: false, message: "Invalid username/email or password." };
}

/**
 * Sends a 6-digit OTP code to the registered email for password recovery.
 */
function sendPasswordResetOTP(usersSheet, otpSheet, identifier) {
  if (!identifier) return { success: false, message: "Please enter your username or email." };

  var userData = usersSheet.getDataRange().getValues();
  var idLower = String(identifier).trim().toLowerCase();
  var targetEmail = "";

  for (var i = 1; i < userData.length; i++) {
    if (String(userData[i][0]).toLowerCase() === idLower || String(userData[i][1]).toLowerCase() === idLower) {
      targetEmail = String(userData[i][1]).trim();
      break;
    }
  }

  if (!targetEmail) {
    return { success: false, message: "Registered account not found." };
  }

  var otpCode = Math.floor(100000 + Math.random() * 900000).toString();
  var expiresAt = new Date(Date.now() + 10 * 60 * 1000).toISOString();

  otpSheet.appendRow([targetEmail, otpCode, expiresAt, false]);

  try {
    MailApp.sendEmail(targetEmail, "ID Card Maker - Password Reset OTP", "Your OTP for resetting your password is: " + otpCode + "\n\nThis OTP will expire in 10 minutes.");
    return { success: true, message: "OTP sent to registered email: " + targetEmail, email: targetEmail };
  } catch (err) {
    return { success: false, message: "Failed to send email OTP: " + err.toString() };
  }
}

/**
 * Resets user password after verifying OTP.
 */
function resetPasswordWithOTP(usersSheet, otpSheet, resetData) {
  if (!resetData || !resetData.email || !resetData.otp || !resetData.newPassword) {
    return { success: false, message: "All fields are required." };
  }

  var otpData = otpSheet.getDataRange().getValues();
  var emailLower = String(resetData.email).trim().toLowerCase();
  var isValidOtp = false;
  var otpRowIndex = -1;

  for (var i = otpData.length - 1; i >= 1; i--) {
    if (String(otpData[i][0]).toLowerCase() === emailLower &&
        String(otpData[i][1]) === String(resetData.otp) &&
        !otpData[i][3]) {
      
      var expires = new Date(otpData[i][2]);
      if (expires.getTime() > Date.now()) {
        isValidOtp = true;
        otpRowIndex = i + 1;
        break;
      }
    }
  }

  if (!isValidOtp) {
    return { success: false, message: "Invalid or expired OTP code." };
  }

  var userData = usersSheet.getDataRange().getValues();
  var userRowIndex = -1;
  for (var j = 1; j < userData.length; j++) {
    if (String(userData[j][1]).toLowerCase() === emailLower) {
      userRowIndex = j + 1;
      break;
    }
  }

  if (userRowIndex > 0) {
    usersSheet.getRange(userRowIndex, 3).setValue(resetData.newPassword);
    otpSheet.getRange(otpRowIndex, 4).setValue(true);
    return { success: true, message: "Password reset successfully! You can now log in." };
  }

  return { success: false, message: "User account not found." };
}

/**
 * Writes text into adjacent cells in chunks to bypass Google Sheets' 50,000 character cell limit.
 */
function writeChunkedPayload(sheet, rowIndex, startCol, text) {
  var CHUNK_SIZE = 40000;
  var chunks = [];
  for (var i = 0; i < text.length; i += CHUNK_SIZE) {
    chunks.push(text.substring(i, i + CHUNK_SIZE));
  }
  if (chunks.length === 0) chunks.push("");

  var maxCols = Math.max(sheet.getLastColumn(), startCol + chunks.length + 5);
  sheet.getRange(rowIndex, startCol, 1, maxCols - startCol + 1).clearContent();
  sheet.getRange(rowIndex, startCol, 1, chunks.length).setValues([chunks]);
}

/**
 * Reads and concatenates chunked text across cells starting at `startColIndex`.
 */
function readChunkedPayload(rowValues, startColIndex) {
  var chunks = [];
  for (var i = startColIndex; i < rowValues.length; i++) {
    if (rowValues[i] !== "" && rowValues[i] !== null && rowValues[i] !== undefined) {
      chunks.push(String(rowValues[i]));
    } else {
      break;
    }
  }
  return chunks.join("");
}

/**
 * Saves or updates a template record using chunked storage for large image payloads.
 */
function saveTemplateRecord(sheet, template, userToken) {
  if (!template || !template.uid) {
    return { success: false, message: "Invalid template payload." };
  }

  var data = sheet.getDataRange().getValues();
  var uidStr = String(template.uid);
  var rowIndex = -1;

  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === uidStr) {
      rowIndex = i + 1;
      break;
    }
  }

  var now = new Date().toISOString();
  var dimensionsStr = JSON.stringify(template.dimensions || {});
  var owner = (userToken && userToken.username) ? userToken.username : "ANONYMOUS";
  var isApproved = (userToken && userToken.isAdmin) ? true : false;

  var payloadObject = {
    backgroundStyles: template.backgroundStyles || {},
    elementsDataset: template.elementsDataset || []
  };
  var payloadStr = JSON.stringify(payloadObject);

  if (rowIndex > 0) {
    var existingOwner = data[rowIndex - 1][3];
    var existingApproved = Boolean(data[rowIndex - 1][4]);

    // Regular users updating an approved template will automatically branch off a personal copy
    if (existingApproved && !isApproved) {
      var newUid = Date.now();
      var newRowIndex = sheet.getLastRow() + 1;
      sheet.getRange(newRowIndex, 1, 1, 6).setValues([[
        String(newUid),
        template.label + " (My Copy)",
        dimensionsStr,
        owner,
        false,
        now
      ]]);
      writeChunkedPayload(sheet, newRowIndex, 7, payloadStr);
      return { success: true, message: "Branch copy saved under your account.", uid: newUid };
    }

    // Direct update
    sheet.getRange(rowIndex, 1, 1, 6).setValues([[
      uidStr,
      template.label || "Untitled Template",
      dimensionsStr,
      existingOwner || owner,
      existingApproved || isApproved,
      now
    ]]);
    writeChunkedPayload(sheet, rowIndex, 7, payloadStr);
    return { success: true, message: "Template changes updated successfully.", uid: template.uid };
  } else {
    // Append new template
    var appendRowIndex = sheet.getLastRow() + 1;
    sheet.getRange(appendRowIndex, 1, 1, 6).setValues([[
      uidStr,
      template.label || "Untitled Template",
      dimensionsStr,
      owner,
      isApproved,
      now
    ]]);
    writeChunkedPayload(sheet, appendRowIndex, 7, payloadStr);
    return { success: true, message: isApproved ? "Approved Public Template published!" : "Template saved successfully.", uid: template.uid };
  }
}

/**
 * Loads template records with re-assembled payload chunks and pagination support.
 */
function loadTemplateRecords(sheet, username, isAdmin, offset, limit) {
  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) {
    return { success: true, approvedTemplates: [], userTemplates: [], totalUserTemplates: 0, hasMore: false };
  }

  var maxCols = Math.max(7, sheet.getLastColumn());
  var data = sheet.getRange(2, 1, lastRow - 1, maxCols).getValues();

  var approvedTemplates = [];
  var userTemplates = [];

  for (var i = data.length - 1; i >= 0; i--) {
    var row = data[i];
    var rawPayload = readChunkedPayload(row, 6);
    
    var backgroundStyles = { color: "#ffffff", imageSource: "none", opacity: "1" };
    var elementsDataset = [];

    if (rawPayload) {
      try {
        var parsedPayload = JSON.parse(rawPayload);
        backgroundStyles = parsedPayload.backgroundStyles || backgroundStyles;
        elementsDataset = parsedPayload.elementsDataset || elementsDataset;
      } catch (e) {}
    }

    var item = {
      uid: row[0],
      label: row[1],
      dimensions: typeof row[2] === 'string' ? JSON.parse(row[2]) : row[2],
      backgroundStyles: backgroundStyles,
      elementsDataset: elementsDataset,
      owner: row[3],
      isApproved: Boolean(row[4]),
      updatedAt: row[5]
    };

    if (item.isApproved) {
      approvedTemplates.push(item);
    } else if (username && String(item.owner).toLowerCase() === String(username).toLowerCase()) {
      userTemplates.push(item);
    }
  }

  var pagedUserTemplates = userTemplates.slice(offset, offset + limit);
  var hasMore = (offset + limit) < userTemplates.length;

  return {
    success: true,
    approvedTemplates: approvedTemplates,
    userTemplates: pagedUserTemplates,
    totalUserTemplates: userTemplates.length,
    hasMore: hasMore
  };
}

/**
 * Deletes a template record by UID.
 */
function deleteTemplateRecord(sheet, uid, username, isAdmin) {
  if (!uid) return { success: false, message: "Missing template identifier." };
  var data = sheet.getDataRange().getValues();

  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(uid)) {
      var isApproved = Boolean(data[i][4]);
      var owner = String(data[i][3]);

      if (isApproved && !isAdmin) {
        return { success: false, message: "Regular users cannot delete Approved Public Templates." };
      }

      if (!isAdmin && String(owner).toLowerCase() !== String(username).toLowerCase()) {
        return { success: false, message: "You can only delete your own created templates." };
      }

      sheet.deleteRow(i + 1);
      return { success: true, message: "Template deleted successfully from backend." };
    }
  }
  return { success: false, message: "Target template record not found." };
}
